    def design_matrix(feats: pd.DataFrame, regions: list[str]) -> pd.DataFrame:
        X = pd.DataFrame(index=feats.index)
        for r in regions[1:]:  # first region is the reference level
            X[f"region={r}"] = (feats.region == r).astype(float)
        X["hour_sin"], X["hour_cos"] = feats.hour_sin, feats.hour_cos
        X["n_active_oblasts"] = feats.n_active_oblasts.astype(float)
        X["log_prev_duration"] = np.log1p(feats.prev_duration_min)
        return X
    
    
    def fit(feats: pd.DataFrame, durations: pd.Series, regions: list[str]) -> tuple[dict, LogNormalAFTFitter]:
        X = design_matrix(feats, regions)
        df = X.assign(duration=durations.clip(lower=0.5).to_numpy())
        aft = LogNormalAFTFitter(penalizer=0.01).fit(df, duration_col="duration")
        mu = aft.params_.loc["mu_"]
        params = {
            "regions": regions,
            "mu_intercept": float(mu["Intercept"]),
            "mu_coef": {c: float(mu[c]) for c in X.columns},
            "log_sigma": float(aft.params_.loc[("sigma_", "Intercept")]),
        }
        return params, aft
    
    
    FAMILIES = {"lognormal": LogNormalAFTFitter, "weibull": WeibullAFTFitter, "loglogistic": LogLogisticAFTFitter}
    
    
    def compare_families(f_tr: pd.DataFrame, d_tr: pd.Series, f_va: pd.DataFrame, d_va: pd.Series,
                         regions: list[str]) -> dict[str, float]:
        """Mean validation log-likelihood (per alert, minutes scale) of each AFT distribution family."""
        # every alert here has an observed end (event = 1); score() needs the column explicitly
        tr = design_matrix(f_tr, regions).assign(duration=d_tr.clip(lower=0.5).to_numpy(), event=1)
        va = design_matrix(f_va, regions).assign(duration=d_va.clip(lower=0.5).to_numpy(), event=1)
        out = {}
        for name, cls in FAMILIES.items():
            m = cls(penalizer=0.01).fit(tr, duration_col="duration", event_col="event")
            out[name] = float(m.score(va, scoring_method="log_likelihood"))
        return out
    
    
    def predict_mu_sigma(params: dict, feats: pd.DataFrame) -> tuple[np.ndarray, float]:
        X = design_matrix(feats, params["regions"])
        coef = pd.Series(params["mu_coef"])
        mu = params["mu_intercept"] + X[coef.index].to_numpy() @ coef.to_numpy()
        return mu, float(np.exp(params["log_sigma"]))
    
    
    def quantile(mu, sigma, q, elapsed=0.0):
        """Duration quantile, optionally conditional on the alert having lasted ``elapsed`` minutes."""
        elapsed = np.asarray(elapsed, dtype=float)
        s_e = np.where(elapsed > 0, norm.sf((np.log(np.maximum(elapsed, 1e-9)) - mu) / sigma), 1.0)
        return np.exp(mu + sigma * norm.ppf(1 - (1 - q) * s_e))
    
    
    def survival_curve(mu: float, sigma: float, grid_min: np.ndarray, elapsed: float = 0.0) -> np.ndarray:
        s = norm.sf((np.log(grid_min) - mu) / sigma)
        if elapsed > 0:
            s = np.minimum(s / norm.sf((np.log(elapsed) - mu) / sigma), 1.0)
            s[grid_min < elapsed] = 1.0
        return s

    ## Part 3: Answers to your questions (no code)
    
    ### Duration model (`training/duration_model.py`)
    - **What it is:** a log-normal accelerated failure time (AFT) model, fitted with lifelines.
      - It assumes `log(duration) ~ Normal(mu(x), sigma)`.
      - `mu(x)` is linear in these inputs: oblast (one-hot), hour (sin/cos), number of oblasts currently on alert, and log of the previous alert's duration.
      - Only the coefficients are exported to JSON. Prediction is closed-form numpy/scipy.
    - **"Time left" output:** the model returns quantiles conditioned on elapsed time: `S(t | lasted e) = S(t) / S(e)`. So the median end moves later as an alert drags on.
    - **Test results:**
    
      | Metric | Model | Baseline |
      |---|---|---|
      | MAE of median | 19.3 min | 19.5 min (per-oblast median) |
      | 80% interval coverage | 81% (well calibrated) | – |
      | C-index | 0.61 (weak ranking) | – |
    
      - sigma = 1.08, which is very wide: the 80% interval spans roughly ÷4 to ×4 around the median.
    - **Pros:**
      - Tiny and fast (about 6 s).
      - Gives a full calibrated distribution.
      - Handles elapsed time exactly.
      - Interpretable.
    - **Cons:**
      - Barely beats the baseline point estimate.
      - Linear in its features.
      - Weibull fit better on validation (−4.31 vs −4.36 log-lik) but isn't used.
      - Durations depend on the threat type (a drone wave vs. a ballistic missile), which the inputs don't include.
      - Cheap upgrades: switch to Weibull, or use LightGBM with a survival objective.
    
    ### LSTM for "P(alert starts in the next hour)"
    - **Pros:**
      - Learns temporal patterns from raw sequences with no hand-made lag features.
      - Can model all 24 oblasts jointly.
    - **Cons, given this project:**
      - The data is small and the regime keeps changing: train starts 2025-10, with about 180k oblast-hours and few independent "attack nights". LSTMs overfit and need careful tuning.
      - Slow on a 4-core CPU: minutes to hours, versus 3 s now.
      - Poorly calibrated without extra work.
      - Hard to explain.
      - Sequence windows make leakage harder to audit.
      - On tabular, event-count problems, gradient boosting with lag features usually matches or beats LSTMs.
    
    ### Better alternatives (cheap compute), in recommended order
    1. **Keep LightGBM** (current). It is already the strong, cheap option; spend the effort on features (weather, Air Force Telegram threat posts).
    2. **Hawkes / self-exciting point process:** models how alerts trigger more alerts, within an oblast and from neighbours. It fits in seconds and is a natural fit for alert cascades. Worth trying as an interpretable rival or as an added feature.
    3. **Discrete-time hazard logistic regression** on the same features. Near-instant, calibrated, and a good sanity check.
    4. If a neural model is ever wanted: a **small TCN or GRU** on the same hourly grid, or a zero-shot foundation model (Chronos/TimesFM). Both cost far more compute than the gain is likely worth here.

    """Model 2: distribution of alert duration (log-normal accelerated failure time model).
    
    log(duration) ~ Normal(mu(x), sigma), so S(t) = 1 - Phi((ln t - mu) / sigma).
    The fitted coefficients are exported to JSON; prediction below needs only numpy/scipy.
    """
    import json
    from pathlib import Path
    
    import numpy as np
    import pandas as pd
    from lifelines import LogLogisticAFTFitter, LogNormalAFTFitter, WeibullAFTFitter
    from lifelines.utils import concordance_index
    from scipy.stats import norm
    
    NUMERIC = ["hour_sin", "hour_cos", "n_active_oblasts", "log_prev_duration"]
    
    
    def design_matrix(feats: pd.DataFrame, regions: list[str]) -> pd.DataFrame:
        X = pd.DataFrame(index=feats.index)
        for r in regions[1:]:  # first region is the reference level
            X[f"region={r}"] = (feats.region == r).astype(float)
        X["hour_sin"], X["hour_cos"] = feats.hour_sin, feats.hour_cos
        X["n_active_oblasts"] = feats.n_active_oblasts.astype(float)
        X["log_prev_duration"] = np.log1p(feats.prev_duration_min)
        return X
    
    
    def fit(feats: pd.DataFrame, durations: pd.Series, regions: list[str]) -> tuple[dict, LogNormalAFTFitter]:
        X = design_matrix(feats, regions)
        df = X.assign(duration=durations.clip(lower=0.5).to_numpy())
        aft = LogNormalAFTFitter(penalizer=0.01).fit(df, duration_col="duration")
        mu = aft.params_.loc["mu_"]
        params = {
            "regions": regions,
            "mu_intercept": float(mu["Intercept"]),
            "mu_coef": {c: float(mu[c]) for c in X.columns},
            "log_sigma": float(aft.params_.loc[("sigma_", "Intercept")]),
        }
        return params, aft
    
    
    FAMILIES = {"lognormal": LogNormalAFTFitter, "weibull": WeibullAFTFitter, "loglogistic": LogLogisticAFTFitter}
    
    
    def compare_families(f_tr: pd.DataFrame, d_tr: pd.Series, f_va: pd.DataFrame, d_va: pd.Series,
                         regions: list[str]) -> dict[str, float]:
        """Mean validation log-likelihood (per alert, minutes scale) of each AFT distribution family."""
        # every alert here has an observed end (event = 1); score() needs the column explicitly
        tr = design_matrix(f_tr, regions).assign(duration=d_tr.clip(lower=0.5).to_numpy(), event=1)
        va = design_matrix(f_va, regions).assign(duration=d_va.clip(lower=0.5).to_numpy(), event=1)
        out = {}
        for name, cls in FAMILIES.items():
            m = cls(penalizer=0.01).fit(tr, duration_col="duration", event_col="event")
            out[name] = float(m.score(va, scoring_method="log_likelihood"))
        return out
    
    
    def predict_mu_sigma(params: dict, feats: pd.DataFrame) -> tuple[np.ndarray, float]:
        X = design_matrix(feats, params["regions"])
        coef = pd.Series(params["mu_coef"])
        mu = params["mu_intercept"] + X[coef.index].to_numpy() @ coef.to_numpy()
        return mu, float(np.exp(params["log_sigma"]))
    
    
    def quantile(mu, sigma, q, elapsed=0.0):
        """Duration quantile, optionally conditional on the alert having lasted ``elapsed`` minutes."""
        elapsed = np.asarray(elapsed, dtype=float)
        s_e = np.where(elapsed > 0, norm.sf((np.log(np.maximum(elapsed, 1e-9)) - mu) / sigma), 1.0)
        return np.exp(mu + sigma * norm.ppf(1 - (1 - q) * s_e))
    
    
    def survival_curve(mu: float, sigma: float, grid_min: np.ndarray, elapsed: float = 0.0) -> np.ndarray:
        s = norm.sf((np.log(grid_min) - mu) / sigma)
        if elapsed > 0:
            s = np.minimum(s / norm.sf((np.log(elapsed) - mu) / sigma), 1.0)
            s[grid_min < elapsed] = 1.0
        return s
    
    
    def evaluate(params: dict, feats: pd.DataFrame, durations: pd.Series, baseline_median: pd.Series) -> dict:
        mu, sigma = predict_mu_sigma(params, feats)

            "dur_mae_median_min": float(np.mean(np.abs(med - y))),
            "dur_mae_baseline_min": float(np.mean(np.abs(base - y))),
            "dur_medae_min": float(np.median(np.abs(med - y))),
            "dur_coverage_80": float(np.mean((y >= lo) & (y <= hi))),

            "dur_mean_log_lik": float(np.mean(norm.logpdf(np.log(np.maximum(y, 0.5)), mu, sigma) - np.log(np.maximum(y, 0.5)))),
            "sigma": sigma,
        }

## 3. The duration model (`training/duration_model.py`)
- **Approach:** it's a log-normal survival model (lifelines' log-normal AFT). It treats the log of the alert's duration as a normal distribution. The centre of that distribution depends on the oblast, the hour of day, how many oblasts are on alert, and how long the previous alert in that oblast lasted. The spread is the same for every alert.
- **Why it suits the task:** it predicts a full range of durations, not just one number. The dashboard shows the median and a 10–90% range, and updates the estimate as the alert goes on. For example, "it has lasted 40 min, so the median end is now X". All of this is a few lines of numpy/scipy, and training takes seconds.
- **Test results:** the error of the median is 19.3 min, against 19.5 min for the per-oblast median, so it's barely better as a single number. The 80% range contains the real duration 81% of the time, which is the useful part.
- **Weaknesses:**
  - The spread is the same for every alert, but a lone-drone alert and a massive missile attack really do differ in how uncertain they are.
  - A Weibull distribution fit the validation data slightly better, but the prediction code only supports log-normal.
  - In Now mode the live feed doesn't give the alert's start time, so the forecast can't account for how long the alert has already lasted.

## 4. LSTM vs. what you have (LightGBM)
**What an LSTM would add:** it learns patterns over time from raw sequences without hand-made features. It can also model all oblasts together and capture how attack waves build up.

**Why it's a poor fit here:**
- **Too little data:** the data has about 9k hours × 24 oblasts, and positive hours are rare. Neural networks overfit on that much, while gradient-boosted trees do well.
- **Compute:** an LSTM is slow to train on a 4-core CPU, needs tuning, and gives different results from run to run. LightGBM trains in about 20 s.
- **Leakage risk:** your current features have tests proving they only use data available at time t. Sequence windows are much easier to get wrong.
- **Recent evidence:** on tabular and event-count data like this, boosted trees with lag features usually match or beat RNNs, as in the M5 competition.

**Alternatives worth considering (all cheap):**
1. **Keep LightGBM** and keep improving features. You're already doing this, and it's the right default.
2. **Hawkes process (self-exciting point process):** each alert raises the chance of more alerts nearby soon after, and that effect fades over time. This matches how attack waves move between oblasts, and it has very few parameters. It works well as an interpretable baseline or as an extra feature.
3. **Logistic regression on the same features:** it's a sanity check. If it gets close to LightGBM, the features matter more than the model.
4. **Pretrained time-series models (Chronos, TimesFM):** they need no training, but they forecast counts rather than event probabilities, and they're heavier to run on a CPU. Only worth a quick comparison.

If you want something more for the start model, a Hawkes baseline makes a good comparison point: about an hour of work, and it trains in seconds.

        e = np.where(a.finished_at.isna(), NEVER, _ns(a.finished_at.fillna(a.started_at)))
        return s, e, a.duration_min.to_numpy()
    
    
    def last_finished(e, dur, t, k=1):
        """Mean log1p duration of the last ``k`` alerts that had ended by t (NaN if none)."""
        fin = e != NEVER
        order = np.argsort(e[fin])
        ends, d = e[fin][order], np.log1p(dur[fin][order])
        j = np.searchsorted(ends, t, side="right")  # number of alerts ended by t
        if len(d) == 0:
            return np.full(len(t), np.nan)
        c = np.concatenate([[0], np.cumsum(d)])
        lo = np.maximum(j - k, 0)
        return np.where(j > 0, (c[j] - c[lo]) / np.maximum(j - lo, 1), np.nan)
    
    
    def n_active_at(per, t):
        return sum(_active_at(s, e, t).astype(int) for s, e, _ in per.values())
    
    
    # ---------------------------------------------------------------- start model
    def start_candidates(alerts, feats, regs) -> pd.DataFrame:
        per = {r: region_arrays(alerts, r) for r in regs}
        t = _ns(feats.t)
        all_s = np.sort(_ns(alerts.started_at))
        ut, inv = np.unique(t, return_inverse=True)
        out = pd.DataFrame(index=feats.index)
        out["national_starts_15m"] = _count_between(all_s, t - 15 * MIN, t)
        out["national_starts_3h"] = _count_between(all_s, t - 3 * HOUR, t)
        out["d_active_1h"] = (n_active_at(per, ut) - n_active_at(per, ut - HOUR))[inv]
        for col in ["prev_duration_log", "min_since_last_start", "nb_starts_3h", "starts_same_hour_7d"]:
            out[col] = np.nan
        for r, idx in feats.groupby("region", observed=True).indices.items():
            s, e, dur = per[r]
            tt = t[idx]
            out.iloc[idx, out.columns.get_loc("prev_duration_log")] = last_finished(e, dur, tt)
            i = np.searchsorted(s, tt, side="left") - 1
            out.iloc[idx, out.columns.get_loc("min_since_last_start")] = np.where(
                i >= 0, np.minimum((tt - s[np.maximum(i, 0)]) / MIN, MAX_GAP_MIN), MAX_GAP_MIN)
            nbs = [n for n in NEIGHBORS.get(r, []) if n in per]
            out.iloc[idx, out.columns.get_loc("nb_starts_3h")] = sum(_count_between(per[n][0], tt - 3 * HOUR, tt) for n in nbs) if nbs else 0
            # same hour-of-day window on each of the previous 7 days (seasonal-naive as a feature)
            out.iloc[idx, out.columns.get_loc("starts_same_hour_7d")] = sum(
                _count_between(s, tt - k * 24 * HOUR, tt - k * 24 * HOUR + HOUR) for k in range(1, 8))
        return out
    
    
    def run_start(alerts, regs, valid_start, test_start, grid_end):
        feats, y = start_feature_frame(alerts, regs, grid_end)
        feats = pd.concat([feats, start_candidates(alerts, feats, regs)], axis=1)
        calm = (feats.active_now == 0).to_numpy()

        return feats, y, score
    
    
    # ---------------------------------------------------------------- duration model
    def duration_candidates(alerts, events, regs, offset_min: float) -> pd.DataFrame:
        """Candidates known at T = start + offset (the alert itself counts as active when offset > 0)."""
        per = {r: region_arrays(alerts, r) for r in regs}
        t0 = _ns(events.started_at)
        T = t0 + int(offset_min * MIN)
        all_s = np.sort(_ns(alerts.started_at))
        fin = alerts[alerts.finished_at.notna()]
        fin_e, fin_d = _ns(fin.finished_at), np.log1p(fin.duration_min.to_numpy())
        order = np.argsort(fin_e)
        fin_e, c = fin_e[order], np.concatenate([[0], np.cumsum(fin_d[order])])
        out = pd.DataFrame(index=events.index)
        out["n_active_T"] = n_active_at(per, T)
        out["wave_starts"] = np.log1p(_count_between(all_s, t0 - 15 * MIN, T))  # starts nationwide, 15 min before .. T
        out["national_starts_1h"] = np.log1p(_count_between(all_s, T - HOUR, T))
        lo, hi = np.searchsorted(fin_e, T - 3 * HOUR, side="right"), np.searchsorted(fin_e, T, side="right")
        out["nat_recent_logdur"] = np.where(hi > lo, (c[hi] - c[lo]) / np.maximum(hi - lo, 1), np.log1p(20))
        for col in ["nb_active_frac", "starts_24h", "gap_log", "prev_dur_mean5"]:
            out[col] = np.nan
        act = {r: _active_at(s, e, T).astype(float) for r, (s, e, _) in per.items()}
        for r, idx in events.groupby("region").indices.items():
            s, e, dur = per[r]
            nbs = [n for n in NEIGHBORS.get(r, []) if n in per]
            out.iloc[idx, out.columns.get_loc("nb_active_frac")] = (sum(act[n][idx] for n in nbs) / len(nbs)) if nbs else 0
            out.iloc[idx, out.columns.get_loc("starts_24h")] = np.log1p(_count_between(s, t0[idx] - 24 * HOUR, t0[idx]))
            fin_r = np.sort(e[e != NEVER])
            j = np.searchsorted(fin_r, t0[idx], side="right") - 1
            gap = np.where(j >= 0, (t0[idx] - fin_r[np.maximum(j, 0)]) / MIN, MAX_GAP_MIN) if len(fin_r) else MAX_GAP_MIN
            out.iloc[idx, out.columns.get_loc("gap_log")] = np.log1p(np.minimum(gap, MAX_GAP_MIN))
            out.iloc[idx, out.columns.get_loc("prev_dur_mean5")] = last_finished(e, dur, t0[idx], k=5)
        return out.fillna({"prev_dur_mean5": np.log1p(20)})
    
    
    def fit_score(X_tr, d_tr, X_va, d_va, entry: float):
        df = X_tr.assign(duration=d_tr, event=1)
        kw = {}
        if entry > 0:
            df["entry"] = entry
            kw["entry_col"] = "entry"
        aft = LogNormalAFTFitter(penalizer=0.01).fit(df, duration_col="duration", event_col="event", **kw)
        p = aft.params_.loc["mu_"]
        mu = p["Intercept"] + X_va[p.drop("Intercept").index].to_numpy() @ p.drop("Intercept").to_numpy()
        sigma = float(np.exp(aft.params_.loc[("sigma_", "Intercept")]))
        y = d_va
        ly = np.log(y)
        # log-likelihood of the duration in minutes, conditional on lasting > entry
        ll = norm.logpdf(ly, mu, sigma) - ly - (norm.logsf((np.log(entry) - mu) / sigma) if entry > 0 else 0)
        med = duration_model.quantile(mu, sigma, 0.5, entry)
        lo, hi = duration_model.quantile(mu, sigma, 0.1, entry), duration_model.quantile(mu, sigma, 0.9, entry)
        return {"loglik": float(np.mean(ll)), "mae": float(np.mean(np.abs(med - y))),
                "medae": float(np.median(np.abs(med - y))), "cov80": float(np.mean((y >= lo) & (y <= hi))), "sigma": sigma}

            print(f"\n=== DURATION, forecast issued at start + {offset} min "

            variants = [("current", [])]
